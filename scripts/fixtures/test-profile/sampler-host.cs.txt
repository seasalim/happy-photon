using System;
using System.IO;
using System.Threading;

internal static class Program
{
    private static int Main(string[] args)
    {
        var arguments = File.ReadAllLines(args[1]);
        using var ready = EventWaitHandle.OpenExisting(arguments[^3]);
        using var release = EventWaitHandle.OpenExisting(arguments[^2]);
        ready.Set();

        if (!release.WaitOne(30000)) return 1;

        File.WriteAllText(arguments[^1], "released");

        return 0;
    }
}
